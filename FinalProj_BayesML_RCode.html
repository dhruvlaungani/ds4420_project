# logistic regression on standardized predictors
manual_prior <- c(
  prior(student_t(4, 0, 5), class = "Intercept"),
  prior(normal(0, 2), class = "b", coef = "ar1_coef"),
  prior(normal(0, 2), class = "b", coef = "ma1_coef"),
  prior(normal(0, 2), class = "b", coef = "log_mean_streams"),
  prior(normal(0, 2), class = "b", coef = "trend_slope"),
  prior(normal(0, 2), class = "b", coef = "volatility"),
  prior(normal(0, 2), class = "b", coef = "early_late_ratio"),
  prior(normal(0, 2), class = "b", coef = "danceability"),
  prior(normal(0, 2), class = "b", coef = "energy"),
  prior(normal(0, 2), class = "b", coef = "valence"),
  prior(normal(0, 2), class = "b", coef = "acousticness"),
  prior(normal(0, 2), class = "b", coef = "loudness"),
  prior(normal(0, 2), class = "b", coef = "speechiness")
)


# fit model
fit <- brm(
  top10 ~ ar1_coef + ma1_coef + log_mean_streams + trend_slope +
          volatility + early_late_ratio +
          danceability + energy + valence +
          acousticness + loudness + speechiness,
  family = bernoulli("logit"),
  data = train,
  prior = manual_prior,
  chains = 4,
  iter = 2000,
  warmup = 500,
  thin = 2,
  cores = 4,
  seed = 42
)

# posterior sigmoid distributions for two example songs
post_draws <- as.data.frame(fit)

sigmoid <- function(x) 1 / (1 + exp(-x))

post_prob <- function(song_row) {
  x <- as.numeric(song_row[predictors])
  lp <- post_draws$b_Intercept +
        post_draws$b_ar1_coef * x[1] +
        post_draws$b_ma1_coef * x[2] +
        post_draws$b_log_mean_streams * x[3] +
        post_draws$b_trend_slope * x[4] +
        post_draws$b_volatility * x[5] +
        post_draws$b_early_late_ratio * x[6] +
        post_draws$b_danceability * x[7] +
        post_draws$b_energy * x[8] +
        post_draws$b_valence * x[9] +
        post_draws$b_acousticness * x[10] +
        post_draws$b_loudness * x[11] +
        post_draws$b_speechiness * x[12]
  sigmoid(lp)
}

probs_high <- post_prob(test[high_idx, ])
probs_low  <- post_prob(test[low_idx,  ])

hist(probs_high,
     main = paste("P(Top 10) —", test$title[high_idx]),
